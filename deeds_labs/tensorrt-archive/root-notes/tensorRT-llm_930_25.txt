#!/usr/bin/env python3
"""
Quantize Gemma3 model to INT4 AWQ for TensorRT-LLM
"""

from autoawq import AutoAWQForCausalLM, AutoTokenizer
from pathlib import Path

MODEL_DIR = Path("/home/james/gemma3_checkpoint_working")   # your FP16 model
OUTPUT_DIR = Path("/home/james/gemma3_awq")                 # new AWQ quantized model

def main():
    print("🔧 Loading model from", MODEL_DIR)
    model = AutoAWQForCausalLM.from_pretrained(
        str(MODEL_DIR),
        trust_remote_code=True,
        device_map="auto",
    )
    tokenizer = AutoTokenizer.from_pretrained(str(MODEL_DIR), trust_remote_code=True)

    print("📉 Quantizing with INT4 AWQ (group size 128, per-channel)...")
    model.quantize(
        quant_config={
            "zero_point": True,
            "q_group_size": 128,
            "w_bit": 4,
            "version": "GEMM"
        }
    )

    print("💾 Saving quantized model to", OUTPUT_DIR)
    model.save_pretrained(str(OUTPUT_DIR))
    tokenizer.save_pretrained(str(OUTPUT_DIR))

    print("✅ Done! Quantized model is ready for TensorRT-LLM.")

if __name__ == "__main__":
    main()
