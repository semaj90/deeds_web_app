  // cpp/tensorrt_llm/kernels/qlora_ops.cu
  template<typename T>
  __global__ void qlora_merge_kernel(
      const int4* base_weights,  // Q4_K_M quantized
      const T* lora_A,           // LoRA down projection
      const T* lora_B,           // LoRA up projection
      T* output,
      float scaling_factor,
      int rank
  ) {
      // Dequantize base -> Apply LoRA -> Requantize
      int tid = blockIdx.x * blockDim.x + threadIdx.x;

      // Custom Q4_K_M dequantization
      float base_val = dequantize_q4_k_m(base_weights[tid]);

      // LoRA addition: base + scale * (A @ B)
      float lora_val = 0.0f;
      for(int r = 0; r < rank; r++) {
          lora_val += lora_A[tid * rank + r] * lora_B[r *
  gridDim.x + blockIdx.x];
      }

      output[tid] = base_val + scaling_factor * lora_val;
  }

  Mixed Attention Pattern (FlashAttention + Sliding Window):

  // cpp/tensorrt_llm/kernels/hybrid_attention.cu
  template<int HEAD_DIM>
  __global__ void hybrid_flash_sliding_attention(
      const half* Q, const half* K, const half* V,
      half* output,
      int window_size,
      bool use_flash_for_global,
      bool use_sliding_for_local

      // Combine Flash for global context + sliding for recent
      if(threadIdx.x < window_size && use_sliding_for_local) {
          // Sliding window attention for recent tokens

  window_size);

      }
  }
