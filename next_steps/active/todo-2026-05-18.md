## Track 8 — XGBoost Hotness from HyperRAG MapReduce Graph Features

```js
// Step 2.5: Graph features from couchdb:pagerank_scores + hg:edge:idx + Neo4j
const pagerankRaw = await redis.get('couchdb:pagerank_scores');
const prMap = pagerankRaw ? JSON.parse(pagerankRaw) : {};

// per-cluster PageRank mean
for (const c of clusterFeatures) {
  const scores = c.members.map(m => prMap[m.stableKey]?.score ?? prMap[m.relativePath] ?? 0);
  c.clusterPageRank = scores.reduce((s,v) => s+v, 0) / (scores.length || 1);
}

// hyperedge counts from hg:edge:idx ZSET
const edgeKeys = await redis.zrange('hg:edge:idx', 0, -1);
for (const key of edgeKeys) {
  const edge = JSON.parse(await redis.get(`hg:edge:${key}`) ?? 'null');
  if (!edge) continue;
  const cA = edge.clusterA ?? edge.cluster_a;
  const cB = edge.clusterB ?? edge.cluster_b;
  if (clusterMap[cA]) { clusterMap[cA].hyperedgeCount++; if (cB !== cA) clusterMap[cA].crossEdges++; }
  if (clusterMap[cB] && cB !== cA) { clusterMap[cB].hyperedgeCount++; clusterMap[cB].crossEdges++; }
}
```
